## Tại Sao Lỗi Này Xảy Ra

Trong file `next.config.js` của bạn, bạn đã cung cấp một cấu hình không hợp lệ cho trường `images`.

## Các Cách Có Thể Để Khắc Phục

Đảm bảo trường `images` của bạn tuân theo hình dạng cấu hình và giá trị được phép:

```js filename="next.config.js"
module.exports = {
  images: {
    // giới hạn 25 giá trị deviceSizes
    deviceSizes: [640, 750, 828, 1080, 1200, 1920, 2048, 3840],
    // giới hạn 25 giá trị imageSizes
    imageSizes: [32, 48, 64, 96, 128, 256, 384],
    // giới hạn 50 giá trị domains (đã lỗi thời)
    domains: [],
    // tiền tố đường dẫn cho Image Optimization API, hữu ích với `loader`
    path: '/_next/image',
    // loader có thể là 'default', 'imgix', 'cloudinary', 'akamai', hoặc 'custom'
    loader: 'default',
    // file với `export default function loader({src, width, quality})`
    loaderFile: '',
    // vô hiệu hóa static imports cho các file hình ảnh
    disableStaticImages: false,
    // minimumCacheTTL tính bằng giây, phải là số nguyên 0 hoặc lớn hơn
    minimumCacheTTL: 14400, // 4 giờ
    // danh sách thứ tự các định dạng hình ảnh tối ưu hóa được chấp nhận (mime types)
    formats: ['image/webp'],
    // cho phép sử dụng hình ảnh SVG nguy hiểm
    dangerouslyAllowSVG: false,
    // thiết lập header Content-Security-Policy
    contentSecurityPolicy: "default-src 'self'; script-src 'none'; sandbox;",
    // thiết lập header Content-Disposition ('inline' hoặc 'attachment')
    contentDispositionType: 'attachment',
    // giới hạn 25 đối tượng
    localPatterns: [],
    // giới hạn 50 đối tượng
    remotePatterns: [],
    // giới hạn 20 số nguyên
    qualities: [25, 50, 75],
    // khi true, mọi hình ảnh sẽ không được tối ưu hóa
    unoptimized: false,
  },
}
```
